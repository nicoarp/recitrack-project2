  async function registrarDeposito() {
    if(typeof window.ethereum === 'undefined') {
      alert('Instala MetaMask para probar!');
      return;
    }
    await ethereum.request({ method: 'eth_requestAccounts' });
    const provider = new ethers.providers.Web3Provider(window.ethereum);
    const signer = provider.getSigner();
    const contract = new ethers.Contract(contractAddress, contractABI, signer);

    const bottleId = document.getElementById("bottleId").value;
    const cantidad = document.getElementById("cantidad").value;
    const eventType = "DepositoLote";
    const description = cantidad + " botellas depositadas";
    const location = "Punto Limpio Central"; // O puedes pedirlo como input

    try {
      const tx = await contract.registerEvent(bottleId, eventType, description, location);
      await tx.wait();
      alert("¡Evento registrado en blockchain!");
    } catch (e) {
      alert("Error al registrar: " + e.message);
    }
  }

  async function consultarHistorial() {
    if(typeof window.ethereum === 'undefined') {
      alert('Instala MetaMask para probar!');
      return;
    }
    await ethereum.request({ method: 'eth_requestAccounts' });
    const provider = new ethers.providers.Web3Provider(window.ethereum);
    const contract = new ethers.Contract(contractAddress, contractABI, provider);

    const bottleId = document.getElementById("bottleId").value;
    const history = await contract.getBottleHistory(bottleId);
    let html = "<ul>";
    for(let i=0; i<history.length; i++) {
      const ev = history[i];
      html += `<li>${ev.eventType} | ${ev.description} | ${ev.location} | ${new Date(ev.timestamp*1000).toLocaleString()} | actor: ${ev.actor}</li>`;
    }
    html += "</ul>";
    document.getElementById("historial").innerHTML = html;
  }
